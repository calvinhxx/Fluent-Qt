ToggleSwitch 开关

开启或关闭一个设置。

本页内容
示例用法API

示例

用法

读取开关状态

开启或关闭一个设置。

#include <FluentQt/BasicInput.h>
using namespace fluent::basicinput;

auto* toggle = new ToggleSwitch(this);
toggle->setOnContent("On");
toggle->setOffContent("Off");
toggle->setAccessibleName("Feature toggle");
auto* status = new Label("State: Off", this);
connect(toggle, &ToggleSwitch::toggled,
        status, [status](bool on) {
            status->setText(on ? "State: On" : "State: Off");
        });
import fluentqt

toggle = fluentqt.ToggleSwitch()
toggle.setOnContent("On")
toggle.setOffContent("Off")
toggle.setAccessibleName("Feature toggle")
status = fluentqt.Label("State: Off")
toggle.toggled.connect(
    lambda on: status.setText("State: On" if on else "State: Off")
)
