CommandBarFlyout 浮动命令栏

在当前内容旁显示的上下文命令。

本页内容
示例用法API

示例

用法

照片的上下文操作

addPrimaryAction() 添加常用命令，addSecondaryAction() 添加更多命令。左键点击时以 Transient 模式显示且不抢焦点，右键时展开完整菜单。

#include <FluentQt/MenusToolbars.h>
using namespace fluent::menus_toolbars;

auto* flyout = new CommandBarFlyout(this);
auto* shareAction = new QAction(
    QIcon(":/icons/share.svg"), "Share", this);
auto* saveAction = new QAction(
    QIcon(":/icons/save.svg"), "Save", this);
auto* deleteAction = new QAction(
    QIcon(":/icons/delete.svg"), "Delete", this);
auto* resizeAction = new QAction(
    QIcon(":/icons/resize.svg"), "Resize", this);
auto* moveAction = new QAction(
    QIcon(":/icons/move.svg"), "Move", this);
flyout->addPrimaryAction(shareAction);
flyout->addPrimaryAction(saveAction);
flyout->addPrimaryAction(deleteAction);
flyout->addSecondaryAction(resizeAction);
flyout->addSecondaryAction(moveAction);

// Proactive click: keep focus on the invoking content.
connect(photo, &QAbstractButton::clicked, flyout, [=] {
    flyout->showAt(
        photo, CommandBarFlyout::ShowMode::Transient);
});

// Reactive context invocation: expand and enter the menu.
photo->setContextMenuPolicy(Qt::CustomContextMenu);
connect(photo, &QWidget::customContextMenuRequested,
        flyout, [=](const QPoint& localPosition) {
    flyout->showAtPoint(
        photo, localPosition,
        CommandBarFlyout::ShowMode::Standard);
});
using fluent::status_info::Toast;
for (QAction* action : flyout->primaryActions() + flyout->secondaryActions()) {
    if (action->isSeparator()) continue;
    connect(action, &QAction::triggered, this, [this, action] {
        QString text = action->text().section('\t', 0, 0).remove('&');
        Toast::showOrUpdateToast(this, "gallery.action-feedback", text + " selected",
            Toast::Informational, 2200, Toast::Top, QMargins(16, 50, 16, 16));
    });
}
import fluentqt

from PySide6.QtCore import QMargins
from PySide6.QtCore import Qt
from PySide6.QtGui import QAction

flyout = fluentqt.CommandBarFlyout()
share_action = QAction("Share")
save_action = QAction("Save")
delete_action = QAction("Delete")
resize_action = QAction("Resize")
move_action = QAction("Move")
flyout.addPrimaryAction(share_action)
flyout.addPrimaryAction(save_action)
flyout.addPrimaryAction(delete_action)
flyout.addSecondaryAction(resize_action)
flyout.addSecondaryAction(move_action)

photo.clicked.connect(
    lambda: flyout.showAt(photo, fluentqt.CommandBarFlyout.ShowMode.Transient)
)
photo.setContextMenuPolicy(Qt.ContextMenuPolicy.CustomContextMenu)
photo.customContextMenuRequested.connect(
    lambda point: flyout.showAtPoint(
        photo,
        point,
        fluentqt.CommandBarFlyout.ShowMode.Standard
    )
)
for action in flyout.primaryActions() + flyout.secondaryActions():
    if action.isSeparator():
        continue
    action.triggered.connect(
        lambda checked=False, item=action: fluentqt.Toast.showOrUpdateToast(
            photo,
            "gallery.action-feedback",
            item.text().split(chr(9), 1)[0].replace("&", "") + " selected",
            fluentqt.Toast.Informational,
            2200,
            fluentqt.Toast.Top,
            QMargins(16, 50, 16, 16)
        )
    )

bool lightDismissConsumesPress() const

为 true 时，CloseOnPressOutside 轻关闭会 「吞掉」这次关闭点击，而非穿透到下方控件——即 ComboBox 下拉的语义（一次点击只关闭弹窗，不会顺带激活下方控件）。 默认 false，保持菜单式行为：关闭点击仍命中其目标。

void setLightDismissConsumesPress(bool consume)
void addLightDismissPassthrough(QWidget* widget)

上面「吞掉」的豁免区域: 落在这些控件内的关闭点击仍会穿透给它们，使同级工具栏/导航栏在弹窗关闭的同时仍可「一次点击」直达。 仅在 lightDismissConsumesPress 为真时有意义。
