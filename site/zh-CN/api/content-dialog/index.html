ContentDialog 内容对话框

需要用户确认时使用的对话框，最多三个按钮。

本页内容
示例用法API

示例

用法

处理按钮结果

设置主要、次要和关闭按钮的文字；对话框关闭后，finished 的结果说明用户点了哪个按钮。

#include <FluentQt/DialogsFlyouts.h>
using namespace fluent::dialogs_flyouts;

auto* dialog = new ContentDialog(window());
dialog->setTitle("Save your work?");
dialog->setContent(new Label("Unsaved changes in Quarterly report will be lost unless you save them."));
dialog->setPrimaryButtonText("Save");
dialog->setSecondaryButtonText("Don't save");
dialog->setCloseButtonText("Cancel");
dialog->setDefaultButton(ContentDialog::Primary);
connect(dialog, &QDialog::finished, this, [this](int result) {
    const QString message = result == ContentDialog::Primary ? "Save selected"
        : result == ContentDialog::Secondary ? "Don't save selected" : "Cancelled";
    Toast::showOrUpdateToast(this, "gallery.action-feedback", message,
        Toast::Informational, 2200, Toast::Top, QMargins(16, 50, 16, 16));
});
connect(dialog, &QDialog::finished, dialog, &QObject::deleteLater);
dialog->open();
import fluentqt

from PySide6.QtCore import QMargins

dialog = fluentqt.ContentDialog(window)
dialog.setTitle("Save your work?")
dialog.setContent(
    fluentqt.Label(
        "Unsaved changes in Quarterly report will be lost unless you save them."
    )
)
dialog.setPrimaryButtonText("Save")
dialog.setSecondaryButtonText("Don't save")
dialog.setCloseButtonText("Cancel")
dialog.setDefaultButton(fluentqt.ContentDialog.Primary)

def finished(result):
    if result == fluentqt.ContentDialog.Primary:
        message = "Save selected"
    elif result == fluentqt.ContentDialog.Secondary:
        message = "Don't save selected"
    else:
        message = "Cancelled"
    fluentqt.Toast.showOrUpdateToast(
        window,
        "gallery.action-feedback",
        message,
        fluentqt.Toast.Informational,
        2200,
        fluentqt.Toast.Top,
        QMargins(16, 50, 16, 16)
    )

dialog.finished.connect(finished)
dialog.finished.connect(dialog.deleteLater)
dialog.open()
