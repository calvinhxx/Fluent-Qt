ScrollView

A scrollable viewport with Fluent scrollbars and optional zoom.

On this page
ExampleUsageAPI

Example

Usage

Content scrolling and zoom

A large content surface can scroll in both directions and zoom within configured bounds.

#include <FluentQt/Scrolling.h>
using namespace fluent::scrolling;

auto* scrollView = new ScrollView(this);
scrollView->setFixedSize(420, 240);
scrollView->setWidget(new ScrollViewDemoCanvas(
    QSize(760, 520), "Content canvas"));
scrollView->setZoomMode(ScrollView::ZoomMode::Enabled);
scrollView->setMinZoomFactor(0.5);
scrollView->setMaxZoomFactor(2.0);
scrollView->setHorizontalScrollBarVisibility(
    ScrollView::ScrollBarVisibility::Auto);
scrollView->setVerticalScrollBarVisibility(
    ScrollView::ScrollBarVisibility::Auto);

connect(zoomInButton, &QPushButton::clicked,
        scrollView, [scrollView] { scrollView->zoomBy(1.25, true); });
connect(zoomOutButton, &QPushButton::clicked,
        scrollView, [scrollView] { scrollView->zoomBy(0.8, true); });
connect(resetButton, &QPushButton::clicked,
        scrollView, [scrollView] { scrollView->resetZoom(true); });
import fluentqt

from PySide6.QtCore import QSize

scroll_view = fluentqt.ScrollView()
scroll_view.setFixedSize(420, 240)
scroll_view.setOwnedContentWidget(
    ScrollViewDemoCanvas(QSize(760, 520), "Content canvas")
)
scroll_view.setZoomMode(fluentqt.ScrollView.ZoomMode.Enabled)
scroll_view.setMinZoomFactor(0.5)
scroll_view.setMaxZoomFactor(2.0)
scroll_view.setHorizontalScrollBarVisibility(
    fluentqt.ScrollView.ScrollBarVisibility.Auto
)
scroll_view.setVerticalScrollBarVisibility(
    fluentqt.ScrollView.ScrollBarVisibility.Auto
)
zoom_in.clicked.connect(lambda: scroll_view.zoomBy(1.25, True))
zoom_out.clicked.connect(lambda: scroll_view.zoomBy(0.8, True))
reset.clicked.connect(lambda: scroll_view.resetZoom(True))
