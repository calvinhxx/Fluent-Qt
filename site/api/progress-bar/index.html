ProgressBar

Shows determinate or indeterminate progress along a line.

On this page
ExampleUsageAPI

Example

Usage

Determinate value

A determinate ProgressBar exposes range, value, ratio, and progress text.

#include <FluentQt/StatusInfo.h>
using namespace fluent::status_info;

auto* progressBar = new ProgressBar(this);
progressBar->setRange(0, 100);
progressBar->setBarWidth(320);

auto* progressBox = new NumberBox(this);
progressBox->setHeader("Progress");
progressBox->setRange(0, 100);
progressBox->setSmallChange(1);
progressBox->setLargeChange(10);
progressBox->setDisplayPrecision(0);
progressBox->setSpinButtonPlacementMode(NumberBox::SpinButtonPlacementMode::Inline);

auto* status = new Label("Progress: 44%", this);
status->setFixedWidth(124);

connect(progressBox, &NumberBox::valueChanged, progressBar, [progressBar, status](double value) {
    progressBar->setValue(std::isfinite(value) ? value : 0.0);
    status->setText(QStringLiteral("Progress: %1%").arg(progressBar->progressText()));
});
progressBox->setValue(44);
import fluentqt

import math

progress_bar = fluentqt.ProgressBar()
progress_bar.setRange(0, 100)
progress_bar.setBarWidth(320)

progress_box = fluentqt.NumberBox()
progress_box.setHeader("Progress")
progress_box.setRange(0, 100)
progress_box.setSmallChange(1)
progress_box.setLargeChange(10)
progress_box.setDisplayPrecision(0)
progress_box.setSpinButtonPlacementMode(
    fluentqt.NumberBox.SpinButtonPlacementMode.Inline
)

status = fluentqt.Label("Progress: 44%")
status.setFixedWidth(124)

def update_progress(value):
    progress_bar.setValue(value if math.isfinite(value) else 0.0)
    status.setText(f"Progress: {progress_bar.progressText()}%")

progress_box.valueChanged.connect(update_progress)
progress_box.setValue(44)
