RadioButton

Lets people select one option from a mutually exclusive group.

On this page
ExampleUsageAPI

Example

Usage

RadioButton group

Sibling radio buttons are mutually exclusive and update the selected option.

#include <FluentQt/BasicInput.h>
using namespace fluent::basicinput;

#include <FluentQt/TextFields.h>

auto* low = new RadioButton("Low", this);
auto* medium = new RadioButton("Medium", this);
auto* high = new RadioButton("High", this);
medium->setChecked(true);
auto* status = new fluent::textfields::Label("Selected: Medium", this);
status->setFluentTypography(Typography::FontRole::Body);
status->setTextColorRole(fluent::textfields::Label::TextColorRole::Primary);
status->setMinimumWidth(status->sizeHint().width());
for (auto* radio : {low, medium, high}) {
    connect(radio, &RadioButton::toggled, status, [radio, status](bool checked) {
        if (checked)
            status->setText(QString("Selected: %1").arg(radio->text()));
    });
}
import fluentqt

status = fluentqt.Label('Selected: Medium')
status.setFluentTypography(fluentqt.FontRole.Body)
status.setTextColorRole(fluentqt.Label.TextColorRole.Primary)
status.setMinimumWidth(status.sizeHint().width())
low = fluentqt.RadioButton('Low')
medium = fluentqt.RadioButton('Medium')
high = fluentqt.RadioButton('High')
medium.setChecked(True)
for radio in (low, medium, high):
    radio.toggled.connect(
        lambda checked, value=radio: status.setText(
        f"Selected: {value.text()}"
        ) if checked else None
    )
