Timeline

Presents caller-owned model rows as a timeline with text, time labels and node states.

On this page
ExampleUsageAPI

Example

Usage

Model-backed timeline

The application supplies text, time labels and node states. Click the button to see a state transition.

#include <FluentQt/Collections.h>
using namespace fluent::collections;

auto* timeline = new Timeline(this);
auto* model = new QStandardItemModel(3, 1, this);
for (int i = 0; i < 3; ++i) {
    model->setData(model->index(i, 0), QString("Node %1").arg(i + 1));
    model->setData(model->index(i, 0), "Optional supporting content supplied by the application.", Timeline::DescriptionRole);
    model->setData(model->index(i, 0), QString("Time label %1").arg(i + 1), Timeline::TimestampRole);
}
model->setData(model->index(1, 0), Timeline::Active, Timeline::StatusRole);
timeline->setModel(model);
connect(changeState, &Button::clicked, model, [model] {
    const auto index = model->index(1, 0);
    model->setData(index, model->data(index, Timeline::StatusRole).toInt() == Timeline::Active
        ? Timeline::Success : Timeline::Active, Timeline::StatusRole);
});
import fluentqt

from PySide6.QtGui import QStandardItemModel

timeline = fluentqt.Timeline()
roles = fluentqt.Timeline.DataRole
states = fluentqt.Timeline.Status
model = QStandardItemModel(3, 1, root)
detail = 'Optional supporting content supplied by the application.'
for i in range(3):
    index = model.index(i, 0)
    model.setData(index, f'Node {i + 1}')
    model.setData(index, detail, roles.DescriptionRole)
    model.setData(
        index,
        f'Time label {i + 1}',
        roles.TimestampRole
    )
model.setData(model.index(1, 0), states.Active, roles.StatusRole)
timeline.setModel(model)
change = fluentqt.Button('Change node state')
def change_state():
    index = model.index(1, 0)
    active = index.data(roles.StatusRole) == states.Active
    state = states.Success if active else states.Active
    model.setData(index, state, roles.StatusRole)
change.clicked.connect(change_state)
