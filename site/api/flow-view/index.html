FlowView

Lays out variable-sized items in wrapping rows, like a tag cloud.

On this page
ExampleUsageAPI

Example

Usage

Network photo flow

Images load asynchronously while the flow layout wraps cards across rows.

#include <FluentQt/Collections.h>
using namespace fluent::collections;

auto* flowView = new FlowView(this);
flowView->setDefaultItemSize(QSize(160, 118));
flowView->setItemDelegate(photoDelegate);
flowView->setModel(model);
flowView->setSelectionMode(FlowView::SelectionMode::Single);
import fluentqt

from PySide6.QtCore import QSize

flow_view = fluentqt.FlowView()
flow_view.setSelectionMode(fluentqt.SelectionMode.Single)
flow_view.setDefaultItemSize(QSize(160, 118))
flow_view.setItemDelegate(photo_delegate)
flow_view.setModel(model)
