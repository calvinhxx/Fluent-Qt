ToggleButton

A button that switches between checked and unchecked states.

On this page
ExampleUsageAPI

Example

Usage

ToggleButton state

A toggle button keeps its checked state visible after invocation.

#include <FluentQt/BasicInput.h>
using namespace fluent::basicinput;

auto* toggle = new ToggleButton("Bold", this);
connect(toggle, &ToggleButton::toggled,
        status, [status](bool checked) {
            status->setText(checked ? "Checked" : "Unchecked");
        });
import fluentqt

toggle = fluentqt.ToggleButton('Bold')
toggle.toggled.connect(
    lambda checked: status.setText(
        "State: Checked" if checked else "State: Unchecked"
    )
)
