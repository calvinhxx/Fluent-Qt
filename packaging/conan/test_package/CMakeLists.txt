cmake_minimum_required(VERSION 3.16)
project(FluentQtConanConsumer LANGUAGES CXX)

find_package(FluentQt CONFIG REQUIRED)
get_target_property(sdk_includes FluentQt::FluentQt INTERFACE_INCLUDE_DIRECTORIES)
find_path(sdk_include_root FluentQt/FluentQt.h PATHS ${sdk_includes} NO_DEFAULT_PATH)
add_library(consumer_headers OBJECT main.cpp)
target_include_directories(consumer_headers PRIVATE "${sdk_include_root}")
target_compile_features(consumer_headers PRIVATE cxx_std_17)
target_compile_options(consumer_headers PRIVATE "$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")
target_link_libraries(consumer_headers PRIVATE Qt6::Widgets)
add_executable(consumer $<TARGET_OBJECTS:consumer_headers>)
target_link_libraries(consumer PRIVATE FluentQt::FluentQt)
