cmake_minimum_required(VERSION 3.21)
project(StudioExportCheck LANGUAGES CXX)
find_package(FluentQt CONFIG REQUIRED)
set(EXPORT_DIR "" CACHE PATH "Directory containing the C++ templates exported by Page Studio")
file(GLOB templates CONFIGURE_DEPENDS "${EXPORT_DIR}/*.cpp")
if(NOT templates)
    message(FATAL_ERROR "No exported Studio templates found in ${EXPORT_DIR}")
endif()

enable_testing()
add_custom_target(studio_exports ALL)
add_library(studio_check_main OBJECT check_page.cpp)
target_compile_features(studio_check_main PUBLIC cxx_std_17)
target_link_libraries(studio_check_main PUBLIC FluentQt::FluentQt)

set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${EXPORT_DIR}/templates.json")
file(READ "${EXPORT_DIR}/templates.json" documents)
foreach(template IN LISTS templates)
    get_filename_component(name "${template}" NAME_WE)
    set(target "studio_export_${name}")
    # Each standalone export supplies its own createStudioPage() factory.
    add_executable(${target} "${template}")
    target_link_libraries(${target} PRIVATE studio_check_main)
    add_dependencies(studio_exports ${target})
    string(JSON child_count LENGTH "${documents}" "${name}" children)
    add_test(NAME ${target} COMMAND ${target} ${child_count} ${name})
    set_tests_properties(${target} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    if(name STREQUAL "aligned-row")
        # The stretch must not satisfy a document that expects one more child widget.
        math(EXPR extra_child "${child_count} + 1")
        add_test(NAME ${target}_rejects_spacer_as_widget COMMAND ${target} ${extra_child})
        set_tests_properties(${target}_rejects_spacer_as_widget PROPERTIES
            ENVIRONMENT "QT_QPA_PLATFORM=offscreen" WILL_FAIL TRUE)
    endif()
endforeach()
